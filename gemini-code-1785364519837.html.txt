        async function parseZipBufferInMemory(zipBuffer) {
            const zip = await JSZip.loadAsync(zipBuffer);
            const fileMap = {};
            let mainModelFileName = null;
            let audioFile = null;

            for (const filename of Object.keys(zip.files)) {
                if (zip.files[filename].dir) continue;
                const file = zip.files[filename];

                if (filename.endsWith('.glb') || filename.endsWith('.gltf')) {
                    mainModelFileName = filename;
                } else if (filename.endsWith('.mp3')) {
                    audioFile = file;
                } else {
                    const blob = await file.async('blob');
                    const blobUrl = URL.createObjectURL(blob);
                    fileMap[filename] = blobUrl;
                    const shortName = filename.split('/').pop();
                    fileMap[shortName] = blobUrl;
                }
            }

            if (audioFile) {
                const audioData = await audioFile.async('arraybuffer');
                if (!audioContext) audioContext = new (window.AudioContext || window.webkitAudioContext)();
                audioBuffer = await audioContext.decodeAudioData(audioData);
            }

            if (mainModelFileName) {
                const modelFile = zip.files[mainModelFileName];
                const modelBuf = await modelFile.async('arraybuffer');

                const manager = new THREE.LoadingManager();
                manager.setURLModifier((url) => {
                    const cleanUrl = url.replace(/^.*[\\\/]/, '');
                    if (fileMap[cleanUrl]) return fileMap[cleanUrl];
                    if (fileMap[url]) return fileMap[url];
                    return url;
                });

                const loader = new GLTFLoader(manager);
                loader.parse(modelBuf, '', (gltf) => {
                    parseAndSetupModel(gltf);
                });
            }
        }

        async function sendModelToOther() {
            if (activeConn && activeConn.open && currentModelBuffer) {
                document.getElementById('conn-status').innerText = "[送信中 0%]";
                document.getElementById('conn-status').style.color = "orange";
                
                const chunkSize = 16 * 1024;
                const totalChunks = Math.ceil(currentModelBuffer.byteLength / chunkSize);

                activeConn.send({
                    type: 'modelStart',
                    totalChunks: totalChunks,
                    fileName: currentModelFileName
                });

                for (let i = 0; i < totalChunks; i++) {
                    if (!activeConn || !activeConn.open) break;

                    const start = i * chunkSize;
                    const end = Math.min(start + chunkSize, currentModelBuffer.byteLength);
                    const chunk = currentModelBuffer.slice(start, end);

                    activeConn.send({
                        type: 'modelChunk',
                        index: i,
                        chunk: chunk
                    });

                    const progress = Math.floor(((i + 1) / totalChunks) * 100);
                    document.getElementById('conn-status').innerText = `[送信中 ${progress}%]`;

                    if (i % 10 === 0) {
                        await new Promise(resolve => setTimeout(resolve, 10));
                    }
                }
                document.getElementById('conn-status').innerText = "[送信完了!]";
                document.getElementById('conn-status').style.color = "green";
            }
        }

        function setupConnection(conn) {
            activeConn = conn;

            const onConnectionOpen = () => {
                document.getElementById('conn-status').innerText = "[接続完了!]";
                document.getElementById('conn-status').style.color = "green";
                if (currentModelBuffer) sendModelToOther();
            };

            if (conn.open) {
                onConnectionOpen();
            } else {
                conn.on('open', onConnectionOpen);
            }

            activeConn.on('data', async (data) => {
                if (!data) return;

                if (data.type === 'aircraftSelect') {
                    const otherSelected = data.mode;
                    const myAssignedMode = (otherSelected === 'A10') ? 'F16' : 'A10';
                    setAircraftChoice(myAssignedMode, false);
                }

                else if (data.type === 'modelStart') {
                    incomingChunks = new Array(data.totalChunks);
                    expectedTotalChunks = data.totalChunks;
                    receivedFileName = data.fileName || 'model.glb';
                    document.getElementById('conn-status').innerText = `[受信中 0%]`;
                    document.getElementById('conn-status').style.color = "orange";
                }

                else if (data.type === 'modelChunk') {
                    incomingChunks[data.index] = data.chunk;
                    const receivedCount = incomingChunks.filter(c => c !== undefined).length;
                    const progress = Math.floor((receivedCount / expectedTotalChunks) * 100);
                    document.getElementById('conn-status').innerText = `[受信中 ${progress}%]`;

                    if (receivedCount === expectedTotalChunks) {
                        document.getElementById('conn-status').innerText = "[モデル構築中...]";
                        
                        let totalByteLength = 0;
                        incomingChunks.forEach(c => {
                            if (c) totalByteLength += (c.byteLength || c.length || 0);
                        });

                        const mergedBuffer = new Uint8Array(totalByteLength);
                        let offset = 0;
                        incomingChunks.forEach(c => {
                            if (c) {
                                const u8 = new Uint8Array(c);
                                mergedBuffer.set(u8, offset);
                                offset += u8.byteLength;
                            }
                        });

                        resetStateForNewModel();
                        const buffer = mergedBuffer.buffer;

                        if (receivedFileName.endsWith('.zip')) {
                            await parseZipBufferInMemory(buffer);
                        } else {
                            const loader = new GLTFLoader();
                            loader.parse(buffer, '', (gltf) => {
                                parseAndSetupModel(gltf);
                            });
                        }
                        document.getElementById('conn-status').innerText = "[同期完了!]";
                        document.getElementById('conn-status').style.color = "green";
                    }
                } 

                else if (data.type === 'telemetry') {
                    if (data.nodes) {
                        for (let name in data.nodes) {
                            const node = modelNodeMap[name];
                            if (node) {
                                const t = data.nodes[name];
                                if (t.p) node.position.set(t.p.x, t.p.y, t.p.z);
                                if (t.q) node.quaternion.set(t.q.x, t.q.y, t.q.z, t.q.w);
                                if (t.s) node.scale.set(t.s.x, t.s.y, t.s.z);
                                if (t.v !== undefined) node.visible = t.v;
                            }
                        }
                    }
                }
            });

            activeConn.on('close', () => {
                document.getElementById('conn-status').innerText = "[切断されました]";
                document.getElementById('conn-status').style.color = "#666";
            });
        }

        document.getElementById('connect-btn').addEventListener('click', () => {

        function parseAndSetupModel(読み込みデータ) {
            現在のモデル = 読み込みデータ.scene;
            シーン.add(現在のモデル);

            アニメーション対象パーツ = [];
            WinMag_BaseObjs = new Array(7).fill(null);
            f16Cameras = [];
            a10Cameras = [];
            modelNodeMap = {};
            nodeLastState = {};

            const 全パーツ名 = [...ハッチ名一覧, ...脚名一覧];

            // F16指定ボーン・パーツ判定ヘルパー (グループ名+ボーン名 または 内包パーツ名)
            function isF16Part(child, boneName) {
                const name = child.name;
                const fullF16Name = `scene_61glb_${boneName}`;
                const f16GroupMatch = child.parent && (child.parent.name === 'scene_61glb' || child.parent.name === '61');
                return name === boneName || name === fullF16Name || (f16GroupMatch && name.includes(boneName));
            }

            現在のモデル.traverse((child) => {
                if (child.name) {
                    modelNodeMap[child.name] = child;
                }

                if (child.name === 'scene_51glb' || child.name === '51') scene51Group = child;
                if (child.name === 'scene_61glb' || child.name === '61') scene61Group = child;

                // カメラの分類
                if (child.isCamera) {
                    child.rotation.order = 'YXZ';
                    if (child.name === 'PerspectiveCamera_7' || child.name === 'PerspectiveCamera_2_1') {
                        f16Cameras.push(child);
                    } else {
                        a10Cameras.push(child);
                    }
                }

                // --- A10 ノード取得 (scene_51glb 内前提) ---
                if (全パーツ名.includes(child.name)) {
                    const originalQuat = child.quaternion.clone();
                    child.quaternion.set(0, 0, 0, 1);
                    アニメーション対象パーツ.push({
                        mesh: child,
                        name: child.name,
                        type: ハッチ名一覧.includes(child.name) ? 'HATCH' : 'GEAR',
                        initQuat: originalQuat,
                        startQuat: new THREE.Quaternion(0, 0, 0, 1)
                    });
                }
                if (child.name === 'キャノピー' && (!scene61Group || !child.ancestors?.includes(scene61Group))) {
                    キャノピーオブジェクト = child;
                    キャノピーInitQuat.copy(child.quaternion);
                    child.quaternion.set(0, 0, 0, 1);
                    キャノピーStartQuat.set(0, 0, 0, 1);
                }
                if (child.name === 'エルロン左') エルロン左 = child;
                if (child.name === 'エルロン右') エルロン右 = child;
                if (child.name === '昇降舵左') 昇降舵左 = child;
                if (child.name === '昇降舵右') 昇降舵右 = child;
                if (child.name === '方向舵左') 方向舵左 = child;
                if (child.name === '方向舵右') 方向舵右 = child;

                if (child.name === 'スロットル右') {
                    スロットル右 = child;
                    スロットル右_InitPos.copy(child.position);
                }
                if (child.name === 'スロットル左') {
                    スロットル左 = child;
                    スロットル左_InitPos.copy(child.position);
                }
                if (child.name === 'エンジン右') エンジン右 = child;
                if (child.name === 'エンジン左') エンジン左 = child;
                if (child.name === '操縦桿') 操縦桿 = child;

                // --- F16 ノード取得 (scene_61glb + ボーン名 / scene_61glb内パーツ) ---
                if (isF16Part(child, '左-エルロン')) f16_左エルロン = child;
                if (isF16Part(child, '右-エルロン')) f16_右エルロン = child;
                if (isF16Part(child, '左-水平尾翼')) f16_左水平尾翼 = child;
                if (isF16Part(child, '右-水平尾翼')) f16_右水平尾翼 = child;
                if (isF16Part(child, '前-操縦桿')) f16_前操縦桿 = child;
                if (isF16Part(child, '後-操縦桿')) f16_後操縦桿 = child;

                if (isF16Part(child, 'キャノピー_1')) {
                    f16_キャノピー = child;
                    f16_キャノピーInitQuat.copy(child.quaternion);
                }

                            mesh: child,

                    if (child.name === bulletNames[i]) {
                        WinMag_BaseObjs[i] = child;
                    }
                }

                if (child.name === 'HUD') {
                    HUDオブジェクト = child;
                    HUDオブジェクト.material = new THREE.MeshBasicMaterial({
                        map: hudTexture,
                        transparent: true,
                        opacity: 0.95,
                        depthWrite: false,
                        side: THREE.FrontSide

                    HUD右オブジェクト = child;
                    HUD右オブジェクト.material = new THREE.MeshBasicMaterial({
                        map: hudTexture,
                        transparent: true,
                        opacity: 0.95,
                        depthWrite: false,
                        side: THREE.FrontSide
                    });
                    HUD右オブジェクト.visible = hudRight有効;
                }

                if (child.name === 'HMD左') {
                    HMD左オブジェクト = child;
                    HMD左オブジェクト.material = new THREE.MeshBasicMaterial({
                        map: hmdLeftTexture,
                        transparent: true,
                        opacity: 0.95,
                        depthWrite: false,
                        side: THREE.FrontSide
                    });
                }

                if (child.name === 'HMD右') {
                    HMD右オブジェクト = child;
                    HMD右オブジェクト.material = new THREE.MeshBasicMaterial({
                        map: hmdRightTexture,
                        transparent: true,
                        opacity: 0.95,
                        depthWrite: false,
                        side: THREE.FrontSide
                    });
                }

                if (child.name === 'HMD') {
                    HMDオブジェクト = child;
                    HMDオブジェクト.material = new THREE.MeshBasicMaterial({
                        map: hmdLeftTexture,
                        transparent: true,
                        opacity: 0.95,
                        depthWrite: false,
                        side: THREE.FrontSide
                    });
                }

                if (child.name === '左画面') {
                    左画面オブジェクト = child;
                    左画面オブジェクト.material = new THREE.MeshBasicMaterial({ map: leftTexture, side: THREE.FrontSide });
                }
                if (child.name === '右画面') {
                    右画面オブジェクト = child;
                    右画面オブジェクト.material = new THREE.MeshBasicMaterial({ map: rightTexture, side: THREE.FrontSide });

            });

            // ノード階層フォールバック検索
            if (!scene61Group) {
                scene61Group = modelNodeMap['scene_61glb'] || modelNodeMap['61'] || modelNodeMap['F16'] || modelNodeMap['f16'];

            if (!scene51Group) {
                scene51Group = modelNodeMap['scene_51glb'] || modelNodeMap['51'] || modelNodeMap['A10'] || modelNodeMap['a10'];

            const size = box.getSize(new THREE.Vector3());

            デフォルトカメラ.position.set(center.x, center.y - size.y * 0.2, center.z + size.z * 0.8);

            applyWeaponVisibility();

        function アニメーション実行() {
            requestAnimationFrame(アニメーション実行);

            const 現在時刻 = performance.now();

            if (現在時刻 - lastScreenUpdateTime > 100) {
                lastScreenUpdateTime = 現在時刻;
                updateScreens();
            }

            // テレメトリ送受信 (全ノードを網羅・完全同期)
            if (activeConn && activeConn.open) {
                if (現在時刻 - lastTelemetryTime > 30) {
                    lastTelemetryTime = 現在時刻;
                    let nodeTransforms = null;

                    if (現在のモデル) {
                        for (let name in modelNodeMap) {
                            const child = modelNodeMap[name];
                            const p = child.position;
                            const q = child.quaternion;
                            const s = child.scale;
                            const v = child.visible;
                            
                            let last = nodeLastState[name];
                            if (!last) {
                                nodeLastState[name] = { 
                                    x: p.x, y: p.y, z: p.z, 
                                    qx: q.x, qy: q.y, qz: q.z, qw: q.w, 
                                    sx: s.x, sy: s.y, sz: s.z,
                                    v: v 
                                };
                                if (!nodeTransforms) nodeTransforms = {};
                                nodeTransforms[name] = {
                                    p: { x: p.x, y: p.y, z: p.z },
                                    q: { x: q.x, y: q.y, z: q.z, w: q.w },
                                    s: { x: s.x, y: s.y, z: s.z },
                                    v: v
                                };
                            } else if (
                                Math.abs(last.x - p.x) > 0.0001 ||
                                Math.abs(last.y - p.y) > 0.0001 ||
                                Math.abs(last.z - p.z) > 0.0001 ||
                                Math.abs(last.qx - q.x) > 0.0001 ||
                                Math.abs(last.qy - q.y) > 0.0001 ||
                                Math.abs(last.qz - q.z) > 0.0001 ||
                                Math.abs(last.qw - q.w) > 0.0001 ||
                                Math.abs(last.sx - s.x) > 0.0001 ||
                                Math.abs(last.sy - s.y) > 0.0001 ||
                                Math.abs(last.sz - s.z) > 0.0001 ||
                                last.v !== v
                            ) {
                                last.x = p.x; last.y = p.y; last.z = p.z;
                                last.qx = q.x; last.qy = q.y; last.qz = q.z; last.qw = q.w;
                                last.sx = s.x; last.sy = s.y; last.sz = s.z;
                                last.v = v;
                                if (!nodeTransforms) nodeTransforms = {};
                                nodeTransforms[name] = {
                                    p: { x: p.x, y: p.y, z: p.z },
                                    q: { x: q.x, y: q.y, z: q.z, w: q.w },
                                    s: { x: s.x, y: s.y, z: s.z },
                                    v: v
                                };
                            }
                        }
                    }

                    if (nodeTransforms) {
                        activeConn.send({
                            type: 'telemetry',
                            nodes: nodeTransforms
                        });
                    }
                }
            }

            // --- 機体操作ロジック ---
            if (selectedAircraftMode === 'F16' && scene61Group) {
                // 【F-16 (scene_61glb)】 ギミック操作

                // A / D : エルロン制御 (Dで 左-エルロン +X, 右-エルロン -X / Aは逆)
                if (キー入力状態.KeyD) {
                    f16_エルロン角度 = Math.min(f16_エルロン角度 + 舵面移動速度, F16_最大舵角);
                } else if (キー入力状態.KeyA) {
                    f16_エルロン角度 = Math.max(f16_エルロン角度 - 舵面移動速度, -F16_最大舵角);
                } else {
                    if (f16_エルロン角度 > 0) f16_エルロン角度 = Math.max(0, f16_エルロン角度 - 舵面移動速度);
                    if (f16_エルロン角度 < 0) f16_エルロン角度 = Math.min(0, f16_エルロン角度 + 舵面移動速度);
                }

                if (f16_左エルロン) f16_左エルロン.rotation.x = f16_エルロン角度;
                if (f16_右エルロン) f16_右エルロン.rotation.x = -f16_エルロン角度;

                // W / S : 水平尾翼制御 (Wで -X, Sで +X)
                if (キー入力状態.KeyS) {
                    f16_昇降舵角度 = Math.min(f16_昇降舵角度 + 舵面移動速度, F16_最大舵角);

                if (f16_左水平尾翼) f16_左水平尾翼.rotation.x = f16_昇降舵角度;
                if (f16_右水平尾翼) f16_右水平尾翼.rotation.x = f16_昇降舵角度;

                        let progress = Math.min(経過 / 3.0, 1.0);

                        let progress = Math.min(経過 / 3.0, 1.0);

                            let baseObj = null;

                            シーン.add(bulletMesh);
                            activeBullets.push({
                                mesh: bulletMesh,
                                velocity: velocityVector,
                                spawnTime: 現在時刻

            for (let i = activeBullets.length - 1; i >= 0; i--) {

            if (selectedAircraftMode === 'A10' && アニメーション状態 !== 'STOPPED' && アニメーション状態 !== 'OPEN') {
                let 経過時間 = (現在時刻 - アニメーション開始時間) / 1000;

                if (アニメーション状態 === 'OPENING_HATCH') {
                    let 進行度 = Math.min(経過時間 / 4.0, 1.0);

                        p.mesh.quaternion.slerpQuaternions(p.startQuat, ゼロ回転, 進行度);

                        p.mesh.quaternion.slerpQuaternions(p.startQuat, ゼロ回転, 進行度);

                    let 進行度 = Math.min(経過時間 / 5.0, 1.0);
                    アニメーション対象パーツ.filter(p => p.type === 'GEAR').forEach(p => {
                        p.mesh.quaternion.slerpQuaternions(p.startQuat, p.initQuat, 進行度);

                        アニメーション開始('CLOSING_HATCH');

                    アニメーション対象パーツ.filter(p => p.type === 'HATCH').forEach(p => {

        function getPylonWeaponIndex(pylonIndex) {
            return 10 - pylonIndex;

        function initWeaponObjects() {
            武装オブジェクトマップ = {};
            if (!現在のモデル) return;

            現在のモデル.traverse((obj) => {
                const name = obj.name;
                const targetPatterns = [
                    /^Mk84_LD([1-9]|10)$/,
                    /^AIM-54([1-9]|10)$/,
                    /^AIM-541$/, /^AIM-542$/, /^AIM-543$/, /^AIM-544$/, /^AIM-545$/, /^AIM-546$/, /^AIM-547$/, /^AIM-548$/, /^AIM-549$/, /^AIM-5410$/,
                    /^AAM-3_PTM1$/, /^AAM-3_PTM2$/, /^AAM-3_PMT1$/, /^AAM-3_PMT2$/
                ];

                const isTarget = targetPatterns.some(pattern => pattern.test(name));
                if (isTarget) {
                    obj.visible = false;
                    武装オブジェクトマップ[name] = obj;
                }
            });
        }

        function applyWeaponVisibility() {
            if (!現在のモデル) return;

            Object.keys(武装オブジェクトマップ).forEach(name => {
                武装オブジェクトマップ[name].visible = false;
            });

            for (let idx = 0; idx < 10; idx++) {
                const pylonName = パイロン名リスト[idx];
                const num = getPylonWeaponIndex(idx);
                const selectedType = パイロン武装状態[pylonName];

                if (selectedType === "Mk84_LD") {
                    const targetName = `Mk84_LD${num}`;
                    if (武装オブジェクトマップ[targetName]) 武装オブジェクトマップ[targetName].visible = true;
                } else if (selectedType === "AIM-54") {
                    const targetName = `AIM-54${num}`;
                    if (武装オブジェクトマップ[targetName]) 武装オブジェクトマップ[targetName].visible = true;
                }
            }

            if (パイロン武装状態["AAM右側_右"] === "AAM-3") {
                if (武装オブジェクトマップ["AAM-3_PTM1"]) 武装オブジェクトマップ["AAM-3_PTM1"].visible = true;
                if (武装オブジェクトマップ["AAM-3_PMT1"]) 武装オブジェクトマップ["AAM-3_PMT1"].visible = true;
            }
            if (パイロン武装状態["AAM右側_左"] === "AAM-3") {
                if (武装オブジェクトマップ["AAM-3_PTM2"]) 武装オブジェクトマップ["AAM-3_PTM2"].visible = true;
                if (武装オブジェクトマップ["AAM-3_PMT2"]) 武装オブジェクトマップ["AAM-3_PMT2"].visible = true;
            }
        }

        function handleWeaponSelectionMenu() {
            if (!武装選択モード有効) {
                武装選択モード有効 = true;
                選択中パイロンインデックス = 0;
            } else {
                選択中パイロンインデックス++;
                if (選択中パイロンインデックス >= パイロン名リスト.length) {
                    武装選択モード有効 = false;
                    選択中パイロンインデックス = -1;
                }
            }
            updateStatusText();
            updateStoresUI();
        }

        function setWeaponForSelectedPylon(type) {
            if (選択中パイロンインデックス < 0 || 選択中パイロンインデックス >= パイロン名リスト.length) return;

            const pylonName = パイロン名リスト[選択中パイロンインデックス];
            setWeaponForPylonByName(pylonName, type);
        }

        function setWeaponForPylonByName(pylonName, type) {
            if ((pylonName !== "AAM右側_右" && pylonName !== "AAM右側_左") && type === "AAM-3") {
                type = "OFF";
            }
            if ((pylonName === "AAM右側_右" || pylonName === "AAM右側_左") && (type === "Mk84_LD" || type === "AIM-54")) {
                type = "OFF";
            }

            パイロン武装状態[pylonName] = type;

            applyWeaponVisibility();
            updateStatusText();
            updateStoresUI();
        }

        function updateStatusText() {
            const statusElem = document.getElementById('status');
            if (!statusElem) return;

            if (!selectedAircraftMode) {
                statusElem.innerHTML = "ファイル選択後、搭乗機体を選択してください。";
                return;

            if (!武装選択モード有効) {
                statusElem.innerHTML = `<b>【自機: ${selectedAircraftMode}】</b> 【C】カメラ切替 / 【矢印キー】カメラ視点移動` +
                    (selectedAircraftMode === 'F16' ? ` / 【W/S】ピッチ / 【A/D】ロール / 【G】降着装置 / 【V】キャノピー` : ``) +
                    (selectedAircraftMode === 'A10' ? ` / 【Q/E】スロットル / 【W/S】ピッチ / 【A/D】ロール / 【< / >】ヨー / 【G】脚ハッチ / 【V】キャノピー / 【@】武装選択 / 【Shift】GAU-8` : ``);
            } else {
                const currentPylon = パイロン名リスト[選択中パイロンインデックス];
                const currentVal = パイロン武装状態[currentPylon];
                statusElem.innerHTML = `<span style="color:red;">【武装選択モード】</span> [1:OFF / 2:Mk84 / 3:AIM-54 / 4:AAM-3]<br>` +
                    `<b>選択中: ${currentPylon} (現在: ${currentVal})</b> | 次のパイロン選択・決定: 【@】`;
            }
        }

        const storesUI = document.getElementById('stores-monitor-ui');
        const pylonContainer = document.getElementById('pylon-list-container');
        const closeStoresBtn = document.getElementById('close-stores-btn');

        function createStoresUIElements() {
            pylonContainer.innerHTML = '';
            パイロン名リスト.forEach((pylonName) => {
                const row = document.createElement('div');
                row.className = 'pylon-row';

                const label = document.createElement('span');
                label.className = 'pylon-label';
                label.innerText = `${pylonName}:`;

                const select = document.createElement('select');
                select.className = 'pylon-select';
                select.id = `select-${pylonName}`;

                let options = [{ value: "OFF", label: "未搭載" }];
                
                if (pylonName === "AAM右側_右" || pylonName === "AAM右側_左") {
                    options.push({ value: "AAM-3", label: "AAM-3" });
                } else {
                    options.push({ value: "Mk84_LD", label: "Mk84_LD" });
                    options.push({ value: "AIM-54", label: "AIM-54" });
                }

                options.forEach(opt => {
                    const optElem = document.createElement('option');
                    optElem.value = opt.value;
                    optElem.innerText = opt.label;
                    select.appendChild(optElem);
                });

                select.addEventListener('change', (e) => {
                    setWeaponForPylonByName(pylonName, e.target.value);
                });

                row.appendChild(label);
                row.appendChild(select);
                pylonContainer.appendChild(row);
            });
        }

        createStoresUIElements();

        function updateStoresUI() {
            if (武装選択モード有効) {
                storesUI.style.display = 'block';
            } else {
                storesUI.style.display = 'none';
            }

            パイロン名リスト.forEach(pylonName => {
                const select = document.getElementById(`select-${pylonName}`);
                if (select) {
                    select.value = パイロン武装状態[pylonName];
                }
            });
        }

        closeStoresBtn.addEventListener('click', () => {
            武装選択モード有効 = false;
            選択中パイロンインデックス = -1;
            updateStatusText();
            updateStoresUI();
        });
